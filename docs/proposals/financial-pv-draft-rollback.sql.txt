-- Restore the original guard. No row/role/balance changes.
CREATE OR REPLACE FUNCTION ab.financial_voucher_guard()RETURNS trigger LANGUAGE plpgsql AS $$BEGIN IF OLD.status IN('Paid','Cancelled')OR(OLD.tenant_id,OLD.company_id,OLD.item_id,OLD.customer_id,OLD.number,OLD.source_key,OLD.amount,OLD.currency,OLD.issued_on,OLD.payload_cipher,OLD.created_by)IS DISTINCT FROM(NEW.tenant_id,NEW.company_id,NEW.item_id,NEW.customer_id,NEW.number,NEW.source_key,NEW.amount,NEW.currency,NEW.issued_on,NEW.payload_cipher,NEW.created_by)OR(OLD.status='Approved'AND(OLD.bank_account_id,OLD.cheque_id)IS DISTINCT FROM(NEW.bank_account_id,NEW.cheque_id))THEN RAISE EXCEPTION 'immutable voucher identity' USING ERRCODE='23514';END IF;RETURN NEW;END $$;
